/**
 * ALMORA review photo upload (Code Snippets, run everywhere).
 *
 * Adds an optional photo field (always stored as a centred 1:1 square) to the product review form. The photo is shrunk and converted to WebP
 * in the browser before sending (light on phones), and again on the server (centred square crop, max 1000px, WebP, quality 80),
 * so no heavy image ever lands in the media library. The review is saved even if there is no photo, and in that
 * case the product photo stays as the review picture (almora-review-cards.php). Email is not asked
 * (almora-review-no-email.php) and the review is published right away; a review with a link is still held.
 * The photo is stored as an attachment ID in comment meta `almora_review_photo` (same key as almora-review-image.php).
 */
if ( ! defined( 'ABSPATH' ) ) {
	return;
}

define( 'ALMORA_RU_MAX_PX', 1000 ); // the stored photo is a centred 1:1 square, at most this many px per side
define( 'ALMORA_RU_QUALITY', 80 );

/* 1. the field, inside the review form on product pages only */
add_filter(
	'comment_form_fields',
	function ( $fields ) {
		if ( ! function_exists( 'is_product' ) || ! is_product() ) {
			return $fields;
		}
		$field = '<p class="comment-form-photo almora-ru">'
			. '<label for="almora_review_file">תמונה (לא חובה)</label>'
			. '<span class="almora-ru__row">'
			. '<label class="almora-ru__btn" for="almora_review_file">בחירת תמונה</label>'
			. '<span class="almora-ru__name" id="almora-ru-name">לא נבחרה תמונה</span>'
			. '<button type="button" class="almora-ru__x" id="almora-ru-x" hidden aria-label="הסרת התמונה">הסרה</button>'
			. '</span>'
			. '<input id="almora_review_file" name="almora_review_file" type="file" accept="image/jpeg,image/png,image/webp,image/heic,image/heif" class="almora-ru__input">'
			. '<span class="almora-ru__prev" id="almora-ru-prev" hidden></span>'
			. '</p>';
		$out = array();
		foreach ( $fields as $k => $v ) {
			if ( 'cookies' === $k ) {
				$out['almora_photo'] = $field;
			}
			$out[ $k ] = $v;
		}
		if ( ! isset( $out['almora_photo'] ) ) {
			$out['almora_photo'] = $field;
		}
		return $out;
	},
	20
);

/* 3. the server side: save the photo as a WebP attachment and link it to the review */
function almora_ru_store( $comment_id ) {
	if ( empty( $_FILES['almora_review_file'] ) || ! is_array( $_FILES['almora_review_file'] ) ) {
		return;
	}
	$f = $_FILES['almora_review_file'];
	if ( empty( $f['tmp_name'] ) || UPLOAD_ERR_OK !== (int) $f['error'] || ! is_uploaded_file( $f['tmp_name'] ) ) {
		return;
	}
	if ( $f['size'] > 12 * MB_IN_BYTES ) {
		return;
	}
	$comment = get_comment( $comment_id );
	if ( ! $comment || 'product' !== get_post_type( (int) $comment->comment_post_ID ) ) {
		return;
	}

	require_once ABSPATH . 'wp-admin/includes/file.php';
	require_once ABSPATH . 'wp-admin/includes/image.php';
	require_once ABSPATH . 'wp-admin/includes/media.php';

	/* only real images, by content, and only these types */
	$check = wp_check_filetype_and_ext( $f['tmp_name'], $f['name'], array( 'jpg|jpeg|jpe' => 'image/jpeg', 'png' => 'image/png', 'webp' => 'image/webp', 'heic' => 'image/heic', 'heif' => 'image/heif' ) );
	if ( empty( $check['type'] ) ) {
		return;
	}
	$info = @getimagesize( $f['tmp_name'] );
	if ( ! $info ) {
		return;
	}

	$editor = wp_get_image_editor( $f['tmp_name'] );
	if ( is_wp_error( $editor ) ) {
		return;
	}
	if ( method_exists( $editor, 'maybe_exif_rotate' ) ) {
		$editor->maybe_exif_rotate();
	}
	/* always a centred 1:1 square, so every review picture looks the same (no 9:16 or panorama) */
	$size = $editor->get_size();
	$side = min( $size['width'], $size['height'] );
	if ( $size['width'] !== $size['height'] ) {
		$editor->crop( (int) floor( ( $size['width'] - $side ) / 2 ), (int) floor( ( $size['height'] - $side ) / 2 ), $side, $side );
	}
	if ( $side > ALMORA_RU_MAX_PX ) {
		$editor->resize( ALMORA_RU_MAX_PX, ALMORA_RU_MAX_PX, false );
	}
	$editor->set_quality( ALMORA_RU_QUALITY );

	$up     = wp_upload_dir();
	$target = trailingslashit( $up['path'] ) . wp_unique_filename( $up['path'], 'review-' . $comment_id . '.webp' );
	$saved  = $editor->save( $target, 'image/webp' );
	if ( is_wp_error( $saved ) || empty( $saved['path'] ) || ! file_exists( $saved['path'] ) ) {
		return;
	}

	$att = wp_insert_attachment(
		array(
			'post_mime_type' => 'image/webp',
			'post_title'     => 'תמונה לביקורת ' . $comment_id,
			'post_status'    => 'inherit',
			'post_content'   => '',
		),
		$saved['path'],
		(int) $comment->comment_post_ID
	);
	if ( is_wp_error( $att ) || ! $att ) {
		return;
	}
	wp_update_attachment_metadata( $att, wp_generate_attachment_metadata( $att, $saved['path'] ) );
	update_comment_meta( $comment_id, 'almora_review_photo', (int) $att );
	if ( function_exists( 'rocket_clean_domain' ) ) {
		rocket_clean_domain();
	}
}
add_action( 'comment_post', 'almora_ru_store', 20, 1 );

/* 4. never let a photo problem lose the review: the form post with a file bigger than the server limit would come empty, so tell it clearly */
add_filter(
	'preprocess_comment',
	function ( $data ) {
		if ( ! empty( $_FILES['almora_review_file']['error'] ) && UPLOAD_ERR_NO_FILE !== (int) $_FILES['almora_review_file']['error'] && UPLOAD_ERR_OK !== (int) $_FILES['almora_review_file']['error'] ) {
			unset( $_FILES['almora_review_file'] );
		}
		return $data;
	},
	1
);
